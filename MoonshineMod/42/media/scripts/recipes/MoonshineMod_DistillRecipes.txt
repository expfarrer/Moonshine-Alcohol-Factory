        timedAction = Making, category = Moonshine,
        Tags = InHandCraft;Moonshine;CanBeDoneFromFloor,
        time = 900,
        NeedToBeLearn = true,
        SkillRequired = MetalWelding:4;Electricity:2,
        xpAward = MetalWelding:25,
        inputs
        {
        item 1 tags[base:hammer] mode:keep,
        item 1 tags[base:saw] mode:keep,
        item 1 [Base.Extinguisher] flags[ItemCount],
        item 1 [Base.AlarmClock2],
        /* A car battery is KeepOnDeplete: flags[ItemCount] drains its whole
           100000 charge but LEAVES the (now flat) battery in the inventory --
           measured "CarBattery1 uses-100000.00" with no count change.  So the
           player already keeps the battery here, and DismantleControllableColumn
           must NOT hand back a second, full one (that was +1 battery per
           build/dismantle cycle).  Deliberately no `mode:destroy`: this recipe
           would then have two destroyed input types, which is the rule-18/33
           crafting-UI hazard, and that is exactly what a headless test cannot
           see. */
        item 1 [Base.CarBattery1;Base.CarBattery2;Base.CarBattery3] flags[ItemCount],
        item 1 [Base.Wire] flags[ItemCount],
        item 5 [Base.ElectronicsScrap],
        item 1 [Base.DuctTape],
        }
        outputs
        {
            item 1 Moonshine.DistillPotColumn,
        }
    }

    recipe Create Distill Pot(I)
    {
        keep [Recipe.GetItemTypes.Hammer],
        keep [Recipe.GetItemTypes.Saw],

        destroy BucketEmpty,
        MetalPipe=2,
		ScrapMetal,
       
	    Water=2,
        Wire,
        Bleach,
		
        destroy Pot,
        destroy SmallSheetMetal,
        destroy DuctTape,
        destroy PropaneTank,
        destroy BeerCanEmpty,

        Result:DistillPotSmall,
        Time:900.0,

         Sound:BreakMetalItem,
         Prop1:Screwdriver,
         Prop2:Source=4,
         AnimNode:Disassemble,
		 
        CanBeDoneFromFloor:true,
		NeedToBeLearn:true,
		OnGiveXP:Recipe.OnGiveXP.MetalWelding25,
		SkillRequired:MetalWelding=1,
		
    }

    /*------------FILL WOOD AND REMOVE WOOD----------------------------------------*/

    recipe Fill Distill(I) with Wood 
    {
        destroy DistillPotSmall,
        TreeBranch=6,
        Result:DistillPotSmallWood,
        Time:400.0,
        CanBeDoneFromFloor: true,
	
        /*Prop2:Source=1,*/
		Prop1:TreeBranch,
        AnimNode:RipSheets,
		
		
    }
	
	
	
	

    recipe Remove Wood from Distill(I)  
    {
        destroy DistillPotSmallWood,
        Result: DistillPotSmall,
        Time: 400.0,
        CanBeDoneFromFloor: true,
		OnCreate:Moonshine.GiveWoodSmall,
		
		 /*Prop2:Source=1,*/
		Prop1:TreeBranch,
        AnimNode:RipSheets,
		
    }

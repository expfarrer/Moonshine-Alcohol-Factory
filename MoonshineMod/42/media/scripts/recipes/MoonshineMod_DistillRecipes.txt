        time = 400,
        inputs

        /* F11/U5: whole buckets.  `item 10` on a base:food HungerChange

        {

            item 1 Base.BucketEmpty,
            /* O2 (round 7): the tarp + tie the player paid at
               Cover{Corn,Potato}MashBucketWithTarp come back here, when the cover
               comes off to load the still.  This used to be
               `OnCreate = RecipeCodeOnCreate.BucketCover`, whose ZombRand(1,7)
               ladder ALWAYS returned Tarp+RubberBand and additionally handed out a
               free extra RubberBand (1-in-6) or a free Base.Button (1-in-6).
               Native output lines instead: deterministic, exactly the refund, and
               they go through getCreatedOutputItems() so they survive the MP
               AddItem sync gap. */
            item 1 Base.Tarp,
            item 1 Base.RubberBand,
        }
    }

    craftRecipe FillDistillWithWater
    {
        timedAction = Making, category = Moonshine,
        Tags = InHandCraft;Moonshine;CanBeDoneFromFloor,
        time = 400,
        inputs
        {
        /* F11/U4, quirks rule 30.  The old shape was a bare
           `-fluid 40.0 [Water]` with NO kept companion container, so it
           could never source the water and never offered -- measured
           UNAVAILABLE with 10 full 10 L buckets AND 10 full water bottles
           in the bag.  40/60/80 L was also four full buckets.
           Now: a dedicated mode:keep water container, a DIFFERENT item
           type from the DistillPot being consumed, and 4.0 L -- one litre
           per serving the filled pot yields.  A `-fluid` line SUMS across
           every kept container matching its input line (F11/U6b), so this
           takes one bucket, or 4 water bottles, or any mix. */
        item 1 [Base.BucketEmpty;Base.WaterBottle] mode:keep,
        -fluid 4.0 [Water],
        item 1 [Moonshine.DistillPotSmall],
        }
        outputs
        {
            item 1 Moonshine.Water_DistillPotSmall,
        }
    }
